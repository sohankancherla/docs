## Using Hooks

<Info>You can only call Hooks at the top of your components</Info>

### [Sharing Data Between Components](https://react.dev/learn/sharing-state-between-components)

Each component that uses a hook has its own state. If you want to share data between components, you can use a hook to store the data in a parent component and pass it down to the child components.

<Tip> For complex state management, use Context API or a state management library like Zustand.</Tip>

```jsx
function MyApp() {
  const [count, setCount] = useState<number>(0);

  function handleClick() {
    setCount(count + 1);
  }

  return (
    <div>
      <h1>Counters that update together</h1>
      <MyButton count={count} onClick={handleClick} />
      <MyButton count={count} onClick={handleClick} />
    </div>
  );
}
```


## [useState](https://react.dev/reference/react/useState)

Use `useState` to remember some information between renders.

```jsx {4, 7, 12}
import { useState } from "react";

function MyButton() {
  const [count, setCount] = useState<number>(0);

  function handleClick() {
    setCount(count + 1); // Triggers a re-render
  }

  return (
    <button onClick={handleClick}>
      Clicked {count} times
    </button>
  );
}
```

### Updating State

<Warning>A common mistake is to assume that the state is updated immediately when the state is changed. Instead, the state is updated on every render. To update the state incrementally, you need to use an <strong>updater function</strong>.</Warning>

```jsx {6}
const [count, setCount] = useState<number>(0);

function handleClick() {
  setCount(count + 1); // sets count to 1
  setCount(count + 1); // sets count to 1
  setCount(prevCount => prevCount + 1); // sets count to 2
}
```

#### Updating Objects

<Warning>To update an object, replace the object with a new object. If the object is mutated, react is unaware of the change and will not trigger a re-render.</Warning>

```jsx {8-10}
function Profile() {
  const [person, setPerson] = useState({
    name: "John",
    age: 30
  });

  function handleChange(e: ChangeEvent<HTMLInputElement>) {
    setPerson({
      [e.target.name]: e.target.value, // Update the state based on the input name
      ...person  // Copy the rest of the object
    });
  }
}
```

#### Updating Arrays

<Warning>Similar to objects, prefer replacing the array instead of mutating it.</Warning>

``` jsx
const [todos, setTodos] = useState<Todo[]>([]);

function handleAddTodo(todo: Todo) {
  setTodos([...todos, todo]);
}

function handleUpdateTodo(todo: Todo) {
  setTodos(todos.map(todoItem => todoItem.id === todo.id ? todo : todoItem));
}

function handleRemoveTodo(todo: Todo) {
  setTodos(todos.filter(todoItem => todoItem.id !== todo.id));
}

function handleInsertTodo(todo: Todo, index: number) {
  setTodos([...todos.slice(0, index), todo, ...todos.slice(index)]);
}

function handleSortTodos() {
  const tempTodos = [...todos];
  tempTodos.sort((a, b) => a.title.localeCompare(b.title));
  setTodos(tempTodos);
}
```

## [useReducer](https://react.dev/reference/react/useReducer)

use `useReducer` as a style choice to consolidate state update logic.

<CodeGroup>
```jsx Todos.tsx
import { useReducer } from "react";
import todosReducer from "./todosReducer";

const initialTodos = []

function Todos() {
  const [todos, dispatch] = useReducer(todosReducer, initialTodos);

  function handleAddTodo(todo: Todo) {
    dispatch({ type: "added", todo });
  }

  function handleChangeTodo(todo: Todo) {
    dispatch({ type: "changed", todo });
  }

  function handleRemoveTodo(todo: Todo) {
    dispatch({ type: "deleted", todo });
  }

  function handleInsertTodo(todo: Todo, index: number) {
    dispatch({ type: "inserted", todo, index });
  }

  function handleSortTodos() {
    dispatch({ type: "sorted" });
  }
}
```

```jsx todosReducer.ts
export function todosReducer(todos: Todo[], action: Action) {
  switch (action.type) {
    case "added": {
      return [...todos, action.todo];
    }
    case "changed": {
      return todos.map(todoItem => todoItem.id === action.todo.id ? action.todo : todoItem);
    }
    case "deleted": {
      return todos.filter(todoItem => todoItem.id !== action.id);
    }
    case "inserted": {
      return [...todos.slice(0, action.index), action.todo, ...todos.slice(action.index)];
    }
    case "sorted": {
      const tempTodos = [...todos];
      tempTodos.sort((a, b) => a.title.localeCompare(b.title));
      return tempTodos;
    }
    default: {
      throw Error("Unknown action: " + action.type);
    }
  }
}
```
</CodeGroup>

## [useContext](https://react.dev/reference/react/useContext)

Use `useContext` to share data between components without passing props

<Tip>Use `useMemo` to cache the context value when passing objects. This prevents unnecessary re-renders of all consuming components.</Tip>

<CodeGroup>
```jsx App.tsx
import { createContext, useState, useMemo } from 'react';

type ThemeContextType = {
  theme: 'light' | 'dark' | 'system';
  setTheme: (theme: 'light' | 'dark' | 'system') => void;
}

export const ThemeContext = createContext<ThemeContextType>({ theme: 'system', setTheme: () => {} });

function App() {
  const [theme, setTheme] = useState<ThemeContextType['theme']>('system');

  // Memoize the context value to prevent unnecessary re-renders
  const value = useMemo(() => ({ theme, setTheme }), [theme]);

  return (
    <ThemeContext value={theme}>
      <Settings />
    </ThemeContext>
  );
}
```

```jsx Settings.tsx {5,10}
import { useContext } from 'react';
import { ThemeContext } from './App';

function Settings() {
  const { theme, setTheme } = useContext(ThemeContext);

  return (
    <div>
      <h1>Settings</h1>
      <p>Theme: {theme}</p>
      <button onClick={() => setTheme('dark')}>Set Dark</button>
    </div>
  );
}
```
</CodeGroup>
